cmake_minimum_required(VERSION 3.5)
set(CROSS_COMPILE "mips-linux-uclibc-" CACHE STRING "Default userspace toolchain prefix")
if(NOT CMAKE_TOOLCHAIN_FILE AND NOT CMAKE_C_COMPILER AND "$ENV{CC}" STREQUAL "")
    set(CMAKE_SYSTEM_NAME Linux)
    set(CMAKE_SYSTEM_PROCESSOR mips)
    set(CMAKE_C_COMPILER "${CROSS_COMPILE}gcc" CACHE FILEPATH "C compiler")
endif()
project(sm18635_gpio C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(sm18635_gpio src/sm18635_gpio.c)
target_include_directories(sm18635_gpio PRIVATE include)
target_compile_options(sm18635_gpio PRIVATE -Wall -Wextra -Werror)
if(CMAKE_CROSSCOMPILING)
    option(BUILD_TESTING "Build host tests" OFF)
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(test_pin_guard tests/test_pin_guard.c)
    target_compile_options(test_pin_guard PRIVATE -Wall -Wextra -Werror)
    add_test(NAME pin_ownership_policy COMMAND test_pin_guard)
    add_executable(test_unbind_policy tests/test_unbind_policy.c)
    target_compile_options(test_unbind_policy PRIVATE -Wall -Wextra -Werror)
    add_test(NAME automatic_unbind_policy COMMAND test_unbind_policy)
    find_package(PythonInterp 3 REQUIRED)
    add_executable(sm18635_gpio_mock src/sm18635_gpio.c tests/device_mock.c)
    target_include_directories(sm18635_gpio_mock PRIVATE include)
    target_compile_options(sm18635_gpio_mock PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(sm18635_gpio_mock "-Wl,--wrap=open,--wrap=write,--wrap=read,--wrap=close,--wrap=ioctl")
    add_test(NAME protocol_and_tx
        COMMAND ${PYTHON_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_gpio.py
                $<TARGET_FILE:sm18635_gpio> $<TARGET_FILE:sm18635_gpio_mock>)
endif()
